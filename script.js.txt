/* ================= PRIX ================= */

function formatPrice(price) {

  return new Intl.NumberFormat(
    "fr-FR",
    {
      style: "currency",
      currency: "EUR"
    }
  ).format(price);

}


/* ================= AFFICHAGE PRODUITS ================= */

function displayProducts(category = "Tous") {

  const filteredProducts =
    category === "Tous"

      ? products

      : products.filter(
          product =>
            product.category === category
        );


  productsContainer.innerHTML =
    filteredProducts.map(product => {

      return `

        <article class="product">

          <div class="product-image">

            ${
              product.badge

              ?

              `<span class="badge">
                ${product.badge}
              </span>`

              :

              ""

            }

          </div>

          <div class="product-info">

            <h3>
              ${product.name}
            </h3>

            <p>
              ${product.category}
            </p>

            <span class="product-price">
              ${formatPrice(product.price)}
            </span>

            <button
              class="add-cart"
              onclick="addToCart(${product.id})"
            >
              AJOUTER
            </button>

          </div>

        </article>

      `;

    }).join("");

}

/* ================= PANIER ================= */

function addToCart(productId) {

  const product =
    products.find(
      item => item.id === productId
    );

  if (!product) return;


  const existing =
    cart.find(
      item => item.id === productId
    );


  if (existing) {

    existing.quantity += 1;

  } else {

    cart.push({

      ...product,

      quantity: 1

    });

  }


  saveCart();

  openCart();

}


function removeFromCart(productId) {

  cart =
    cart.filter(
      item => item.id !== productId
    );

  saveCart();

}


function changeQuantity(
  productId,
  amount
) {

  const item =
    cart.find(
      product => product.id === productId
    );

  if (!item) return;


  item.quantity += amount;


  if (item.quantity <= 0) {

    removeFromCart(productId);

    return;

  }


  saveCart();

}


function saveCart() {

  localStorage.setItem(
    "alphaFashionCart",
    JSON.stringify(cart)
  );

  displayCart();

}


/* ================= AFFICHAGE PANIER ================= */

function displayCart() {

  const quantity =
    cart.reduce(
      (total, product) =>
        total + product.quantity,
      0
    );


  cartCount.textContent =
    quantity;


  if (cart.length === 0) {

    cartItems.innerHTML = `

      <div class="empty-cart">

        Votre panier est vide.

        <br><br>

        Découvrez notre collection
        et ajoutez vos articles favoris.

      </div>

    `;

    cartTotal.textContent =
      formatPrice(0);

    return;

  }


  cartItems.innerHTML =
    cart.map(product => {

      return `

        <div class="cart-product">

          <div class="cart-product-image"></div>

          <div>

            <h4>
              ${product.name}
            </h4>

            <p>
              ${product.category}
              ·
              ${formatPrice(product.price)}
            </p>

            <button
              class="remove-product"
              onclick="removeFromCart(${product.id})"
            >
              Supprimer
            </button>

            <div style="margin-top:8px">

              <button
                onclick="changeQuantity(${product.id}, -1)"
              >
                −
              </button>

              <strong style="margin:0 8px">
                ${product.quantity}
              </strong>

              <button
                onclick="changeQuantity(${product.id}, 1)"
              >
                +
              </button>

            </div>

          </div>

          <strong>
            ${formatPrice(
              product.price *
              product.quantity
            )}
          </strong>

        </div>

      `;

    }).join("");


  const total =
    cart.reduce(
      (sum, product) =>
        sum +
        product.price *
        product.quantity,
      0
    );


  cartTotal.textContent =
    formatPrice(total);

}


/* ================= OUVERTURE PANIER ================= */

  cartPanel.classList.add("active");

  overlay.classList.add("active");

}


function closeCart() {

  cartPanel.classList.remove("active");

  overlay.classList.remove("active");

}

/* ================= RECHERCHE ================= */

const searchModal =
  document.getElementById("searchModal");

const searchInput =
  document.getElementById("searchInput");

const searchResults =
  document.getElementById("searchResults");


document
  .getElementById("searchButton")
  .addEventListener(
    "click",
    () => {

      searchModal.classList.add("active");

      setTimeout(
        () =>
          searchInput.focus(),
        100
      );

    }
  );


document
  .getElementById("closeSearch")
  .addEventListener(
    "click",
    () =>
      searchModal.classList.remove(
        "active"
      )
  );


searchInput.addEventListener(
  "input",
  event => {

    const query =
      event.target.value
        .toLowerCase()
        .trim();


    if (!query) {

      searchResults.innerHTML = "";

      return;

    }


    const results =
      products.filter(
        product =>
          `${product.name} ${product.category}`
            .toLowerCase()
            .includes(query)
      );


    if (!results.length) {

      searchResults.innerHTML = `

        <p class="search-result">
          Aucun produit trouvé.
        </p>

      `;

      return;

    }


    searchResults.innerHTML =
      results.map(product => `

        <p class="search-result">

          <strong>
            ${product.name}
          </strong>

          — ${formatPrice(product.price)}

        </p>

      `).join("");

  });
